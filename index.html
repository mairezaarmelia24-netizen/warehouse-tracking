const SHEET_NAME = "ACTIVITY_TRACKING"; // ===================================================== // DO GET // ===================================================== function doGet(e) { // API dari website GitHub if (e && e.parameter && e.parameter.api === "processScan") { const callback = e.parameter.callback || "callback"; const data = { operator: e.parameter.operator || "", customerID: e.parameter.customerID || "", transactionID: e.parameter.transactionID || "", department: e.parameter.department || "", activity: e.parameter.activity || "", quantity: e.parameter.quantity || "", action: e.parameter.action || "" }; const result = processScan(data); return ContentService .createTextOutput( callback + "(" + JSON.stringify(result) + ")" ) .setMimeType( ContentService.MimeType.JAVASCRIPT ); } return HtmlService .createHtmlOutputFromFile("Index") .setTitle("Warehouse Activity Tracking"); } // ===================================================== // PROCESS SCAN // ===================================================== function processScan(data) { const sheet = SpreadsheetApp .getActiveSpreadsheet() .getSheetByName(SHEET_NAME); if (!sheet) { return { success: false, type: "ERROR", message: "Sheet ACTIVITY_TRACKING tidak ditemukan." }; } // =================================================== // DATA // =================================================== const operator = String(data.operator || "").trim(); const customerID = String(data.customerID || "").trim(); const activity = String(data.activity || "").trim(); const quantity = String(data.quantity || "").trim(); const action = String(data.action || "").trim().toUpperCase(); // =================================================== // VALIDASI // =================================================== if (!operator) { return { success: false, type: "ERROR", message: "Nama Operator belum diisi." }; } if (!customerID) { return { success: false, type: "ERROR", message: "Customer ID belum dipilih." }; } if (!activity) { return { success: false, type: "ERROR", message: "Activity belum dipilih." }; } if ( action !== "START" && action !== "END" ) { return { success: false, type: "ERROR", message: "QR harus memiliki START atau END." }; } // =================================================== // DATA SHEET // =================================================== const rows = sheet .getDataRange() .getValues(); const now = new Date(); // =================================================== // START // =================================================== if (action === "START") { // ----------------------------------------------- // Cek apakah masih ada pekerjaan Running // ----------------------------------------------- for ( let i = rows.length - 1; i >= 1; i-- ) { const rowCustomer = String(rows[i][2]).trim(); const rowActivity = String(rows[i][4]).trim(); const rowStatus = String(rows[i][9]).trim(); const rowOperator = String(rows[i][10]).trim(); if ( rowCustomer === customerID && rowActivity === activity && rowStatus === "Running" && rowOperator === operator ) { return { success: false, type: "ALREADY_RUNNING", message: "Activity ini masih Running." }; } } // ----------------------------------------------- // Buat Transaction ID otomatis // ----------------------------------------------- const transactionID = generateTransactionID( activity, sheet ); // ----------------------------------------------- // Tracking ID // ----------------------------------------------- const lastRow = sheet.getLastRow(); const trackingID = "TRK-" + String(lastRow) .padStart(3, "0"); // ----------------------------------------------- // Department // ----------------------------------------------- const department = activity.toUpperCase(); // ----------------------------------------------- // Tambahkan data // ----------------------------------------------- sheet.appendRow([ trackingID, transactionID, customerID, department, activity, now, "", "", "", "Running", operator ]); return { success: true, type: "START", trackingID: trackingID, transactionID: transactionID }; } // =================================================== // END // =================================================== if (action === "END") { // ----------------------------------------------- // Quantity wajib diisi // ----------------------------------------------- if (quantity === "") { return { success: false, type: "NEED_QUANTITY", message: "Isi Quantity terlebih dahulu." }; } // ----------------------------------------------- // Cari pekerjaan Running // ----------------------------------------------- for ( let i = rows.length - 1; i >= 1; i-- ) { const rowCustomer = String(rows[i][2]).trim(); const rowActivity = String(rows[i][4]).trim(); const rowStatus = String(rows[i][9]).trim(); const rowOperator = String(rows[i][10]).trim(); if ( rowCustomer === customerID && rowActivity === activity && rowStatus === "Running" && rowOperator === operator ) { // ------------------------------------------- // Start Time // ------------------------------------------- const startTime = new Date(rows[i][5]); // ------------------------------------------- // Duration // ------------------------------------------- const duration = Math.round( (now - startTime) / 60000 ); // ------------------------------------------- // Update Finish // ------------------------------------------- sheet .getRange(i + 1, 7) .setValue(now); sheet .getRange(i + 1, 8) .setValue(duration); sheet .getRange(i + 1, 9) .setValue(quantity); sheet .getRange(i + 1, 10) .setValue("Completed"); return { success: true, type: "FINISH", transactionID: rows[i][1], duration: duration }; } } // ----------------------------------------------- // Tidak menemukan Running // ----------------------------------------------- return { success: false, type: "NO_RUNNING", message: "Tidak ditemukan pekerjaan Running untuk Activity ini." }; } } // ===================================================== // GENERATE TRANSACTION ID // ===================================================== function generateTransactionID( activity, sheet ) { const prefixMap = { "Inbound": "INB", "Inventory": "INV", "Outbound": "OUT", "Auger": "AUG", "Slicing": "SLC", "Rebox": "RBX" }; const prefix = prefixMap[activity] || "TRX"; const date = Utilities.formatDate( new Date(), Session.getScriptTimeZone(), "yyyyMMdd" ); const rows = sheet.getDataRange().getValues(); let count = 0; for ( let i = 1; i < rows.length; i++ ) { const transaction = String(rows[i][1] || ""); if ( transaction.indexOf( prefix + "-" + date ) === 0 ) { count++; } } return ( prefix + "-" + date + "-" + String(count + 1) .padStart(3, "0") ); }
